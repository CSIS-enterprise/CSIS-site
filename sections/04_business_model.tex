\section{Business Model Canvas da CSIS}

\subsection*{Parceiros-Chave}
\begin{itemize}
  \item \textbf{Plataformas Técnicas e Canais de Distribuição:} GitHub (hospedagem rigorosa e controle de \textit{Issues e PRs}), Discord (gestão e engajamento da comunidade), YouTube e plataformas de venda (Hotmart ou similares) para os cursos.
  \item \textbf{Órgãos de Classe e Entidades Representativas:} OABs locais, instituições de perícia digital, Câmaras de Comércio e associações de PMEs, fundamentais como chancela de credibilidade e ponte primária de aproximação ao mercado B2B jurídico.
\end{itemize}

\subsection*{Atividades-Chave}
\begin{itemize}
  \item \textbf{Design e Gestão da Esteira de \textit{Bounties}:} Criação sistemática de missões, gerência de SLAs, revisão do processo de entregas (\textit{Pull Requests}) e auditoria estrita (\textit{Quality Assurance}) dos laudos gerados pelos alunos.
  \item \textbf{Gestão da Comunidade de Elite:} Triagem dos perfis dos talentos, organização das hierarquias e níveis de acesso (arquitetura de \textit{freelancers}), e gerenciamento financeiro das recompensas.
  \item \textbf{Desenvolvimento Curricular:} Definição, curadoria e gravação contínua das trilhas de formação técnica que abastecem o ecossistema.
  \item \textbf{Motor Acadêmico (\textit{Flywheel} de Marketing):} Produção audiovisual intensa e focada na higienização de laudos corporativos anonimizados e transformação dos mesmos em tutoriais de fortíssima autoridade.
  \item \textbf{Relacionamento Corporativo B2B:} Prospecção de casos, modelagem processual para litígios e relacionamento direto na frente de \textit{Customer Success} com o mercado corporativo.
\end{itemize}

\subsection*{Proposta de Valor}
\begin{itemize}
  \item \textbf{Para o Aluno B2C (O Talento):} "Aceleração por imersão prática: Formação técnica aliada à criação efetiva de portfólio de laudos reais e retorno financeiro por cada entrega auditada (\textit{Bounties}). A transição concreta de estudante a prestador de serviço chancelado."
  \item \textbf{Para o Corporativo e Jurídico B2B (O Cliente):} "Tranquilidade e sigilo impenetráveis: Laudos periciais e testes de segurança de padrão-ouro, elaborados numa linguagem executiva para diretorias e tribunais, em um tempo recorde e por uma fatia do alto custo das firmas tradicionais — modelo possibilitado pela esteira de \textit{crowdsourcing} revisada rigorosamente pelo \textit{Expert} sênior."
\end{itemize}

\subsection*{Recursos-Chave}
\begin{itemize}
  \item \textbf{Capital Intelectual Proprietário:} A formidável arquitetura metodológica da "Esteira Forense de \textit{Bounties}", os processos de auditoria (QA) e a originalidade dos laboratórios de rede ensinados no treinamento.
  \item \textbf{O Arquiteto Operacional (Fundador Solopreneur):} Sua densidade curricular, vastas horas de perícia executadas na prática, reputação pessoal e credenciais globais.
  \item \textbf{O Hub de Mão de Obra:} A rede inesgotável e escalável de alunos competindo pelos \textit{freelas}.
  \item \textbf{Ferramental Tático:} Laboratórios para reprodução dos \textit{pentests}, infraestrutura cloud/servidores elásticos e licenças valiosas do mercado corporativo (Ex: softwares forenses homologados).
\end{itemize}

\subsection*{Relacionamento com Clientes}
\begin{itemize}
  \item \textbf{Com O Aluno (B2C):} Aproximação horizontal mediada por uma mentoria tática densa. Cultura focada em comunidade, baseada puramente na meritocracia técnica de entrega no GitHub e no acolhimento acolhedor de novatos rumo ao alto rendimento.
  \item \textbf{Com a Empresa Cliente (B2B):} Relacionamento Vertical (\textit{VIP} - High Ticket). Extremamente pautado em \textit{Accountability}, blindagem judiciária, confidencialidade total e contato estrito unicamente com a figura do consultor/Sênior. A "tradução" do mundo tecnológico superando o \textit{gap} de compreensão do cliente legal.
\end{itemize}

\subsection*{Segmentos de Clientes}
\begin{itemize}
  \item \textbf{O Estudante/Operador (B2C):} Aspirantes ao mercado de Cibersegurança que buscam transição de vida. Jogadores focados em desafios (\textit{Capture The Flag}) que procuram monetizar as habilidades lúdicas. Estão ávidos por portfólio tático.
  \item \textbf{A Instituição Necessitada (B2B):} Pequenas e Médias Empresas (PMEs) enfrentando vazamentos sistêmicos, Ransomware, ou demandando conformidade protetiva; Escritórios Jurídicos lutando contra gargalos investigativos urgentes carentes de perícia digital assinada e confiável.
\end{itemize}

\subsection*{Canais}
\begin{itemize}
  \item \textbf{Canais de Marketing Orgânico de Autoridade:} YouTube ocupando função de tração educacional/conversão e forte portfólio público. LinkedIn executivo tracionando alianças B2B.
  \item \textbf{Canais de Gestão:} Discord atua como "quadro corporativo vivo", e o GitHub funciona como esteira nevrálgica de produção do escritório virtual.
  \item \textbf{Canais Institucionais:} OAB regional, alianças presenciais e conselhos empresariais com potencial de networking imediato para indicações.
\end{itemize}

\subsection*{Fontes de Receita}
\begin{itemize}
  \item \textbf{Produtos Educacionais Base:} Venda recorrente e exponencial das Formações e das Três Camadas Estudantis ("Childrens, Basic e Advanced"), gerando o volume formidável de caixa estrutural mensal e subsidiando todas as premiações da rede \textit{freelance} interna.
  \item \textbf{Alta Consultoria Forense:} Assinatura de laudos, serviços de testagem e auditoria (Receitas isoladas por projeto atreladas diretamente a demandas judiciais ou empresariais exclusivas, constituindo alto tíquete de margem líquida).
\end{itemize}

\subsection*{Estrutura de Custos}
\begin{itemize}
  \item \textbf{Variáveis atrelados ao Custo (COGS):} Provisão para pagamentos por liquidação da Missões na mão dos prestadores alunos. Recompensas financeiras (Ponto cardeal e blindagem do modelo contra altos passivos CLT).
  \item \textbf{Ferramentas e \textit{Stack}:} Hospedagem Cloud (\textit{AWS/Hosting}), plataformas essenciais em nuvem, e pagamentos anuais de licenças forenses irrenunciáveis (necessárias para auditoria final do processo de laudo).
  \item \textbf{Operação e Captação:} Gastos contidos em tráfego para aceleração massiva da formação e custos de edição para as peças audiovisuais \textit{Deep-Dive}.
\end{itemize}
